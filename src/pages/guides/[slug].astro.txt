---
import Base from "../../layouts/Base.astro";
import data from "../../data/articles.json";
import { marked } from "marked";

export function getStaticPaths() {
  return data.articles.map((article) => ({
    params: { slug: article.slug },
    props: { article },
  }));
}

const { article } = Astro.props;
const { articles } = data;

const SITE = "https://typemetrics.net";
const canonical = `${SITE}/guides/${article.slug}`;

const related = (article.relatedSlugs || [])
  .map((s) => articles.find((a) => a.slug === s))
  .filter(Boolean);

const faqs = article.faqs || [];

const bodyHtml = marked.parse(article.body || "");
const formattedDate = article.datePublished
  ? new Date(article.datePublished).toLocaleDateString("en-US", {
      year: "numeric",
      month: "long",
      day: "numeric",
    })
  : null;

const articleJsonLd = {
  "@context": "https://schema.org",
  "@type": "Article",
  headline: article.title,
  description: article.metaDescription || article.excerpt || "",
  datePublished: article.datePublished,
  author: { "@type": "Organization", name: "TypeMetrics" },
  publisher: { "@type": "Organization", name: "TypeMetrics" },
  mainEntityOfPage: { "@type": "WebPage", "@id": canonical },
};
const breadcrumbJsonLd = {
  "@context": "https://schema.org",
  "@type": "BreadcrumbList",
  itemListElement: [
    { "@type": "ListItem", position: 1, name: "Home", item: `${SITE}/` },
    { "@type": "ListItem", position: 2, name: "Guides", item: `${SITE}/guides` },
    { "@type": "ListItem", position: 3, name: article.title, item: canonical },
  ],
};

const jsonLd = [articleJsonLd, breadcrumbJsonLd];
if (faqs.length > 0) {
  jsonLd.push({
    "@context": "https://schema.org",
    "@type": "FAQPage",
    mainEntity: faqs.map((f) => ({
      "@type": "Question",
      name: f.q,
      acceptedAnswer: { "@type": "Answer", text: f.a },
    })),
  });
}

const pageTitle = article.seoTitle || `${article.title} — TypeMetrics`;
---
<Base
  title={pageTitle}
  description={article.metaDescription || article.excerpt || ""}
  canonical={canonical}
  jsonLd={jsonLd}
>
  <nav class="crumbs">
    <a href="/">Home</a><span>›</span>
    <a href="/guides">Guides</a><span>›</span>
    <span>{article.title}</span>
  </nav>

  <article>
    <h1>{article.title}</h1>
    <div class="meta">
      {article.readingTime && <span>{article.readingTime} min read</span>}
      {formattedDate && <span>{formattedDate}</span>}
    </div>
    <div set:html={bodyHtml} />
  </article>

  {faqs.length > 0 && (
    <section class="faq">
      <h2>Frequently asked questions</h2>
      {faqs.map((f) => (
        <div class="faq-item">
          <h3>{f.q}</h3>
          <p>{f.a}</p>
        </div>
      ))}
    </section>
  )}

  <div class="test-cta">
    <p>Put it into practice — measure your WPM and accuracy right now.</p>
    <a class="cta" href="/">Take the Typing Test</a>
  </div>

  {related.length > 0 && (
    <section class="related">
      <h2>Related guides</h2>
      <div class="cards">
        {related.map((r) => (
          <a class="card" href={`/guides/${r.slug}`}>
            <h3>{r.title}</h3>
            <p>{r.excerpt}</p>
          </a>
        ))}
      </div>
    </section>
  )}

  <style>
    .faq {
      margin-top: 48px;
      padding-top: 28px;
      border-top: 1px solid var(--border);
    }
    .faq-item { margin-bottom: 20px; }
    .faq-item h3 { margin: 0 0 4px; font-size: 16px; }
    .faq-item p { margin: 0; color: var(--muted); font-size: 15px; }
  </style>
</Base>
